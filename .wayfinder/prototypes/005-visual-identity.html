-- Ferret query: find active users with recent orders SELECT u.name, u.email, COUNT(o.id) AS order_count, MAX(o.created_at) AS last_order, SUM(o.total)::numeric(10,2) AS lifetime_value FROM users u INNER JOIN orders o ON o.user_id = u.id WHERE u.active = true AND o.created_at >= NOW() - INTERVAL '30 days' GROUP BY u.id, u.name, u.email HAVING COUNT(o.id) > 3 ORDER BY lifetime_value DESC LIMIT 50;
